// 由 scripts/build-apps-script.mjs 與共用資料模型合成 Code.gs。
// 先設定指令碼屬性 SPREADSHEET_ID、API_TOKEN（至少 32 字元）。
const SHEET_HEADERS = {
  Records: ['日期','班別','單位','狀態','上班人力','住院病人數','當班住院','出院','轉入','轉出','病危','死亡','填報內容 JSON','版本','更新時間'],
  Openings: ['單位','日期','班別','期初住院人數'],
  Capacities: ['單位','生效日期','床數','版本','更新時間','調整原因'],
  Audit: ['時間','操作','對象','操作者','調整前 JSON','調整後 JSON']
};
function settings_() {
  const props = PropertiesService.getScriptProperties();
  const id = props.getProperty('SPREADSHEET_ID'), token = props.getProperty('API_TOKEN');
  if (!id || !token || token.length < 32) throw new Error('請設定 SPREADSHEET_ID 與至少 32 字元的 API_TOKEN');
  return { id: id, token: token };
}
function setup() {
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    const cfg = settings_(), book = SpreadsheetApp.openById(cfg.id);
    for (const name in SHEET_HEADERS) {
      const sheet = book.getSheetByName(name) || book.insertSheet(name);
      sheet.getRange(1, 1, 1, SHEET_HEADERS[name].length).setValues([SHEET_HEADERS[name]]).setBackground('#eeeeee').setFontWeight('bold');
      sheet.setFrozenRows(1);
    }
    let internal = book.getSheetByName('_State');
    if (!internal) internal = book.insertSheet('_State');
    if (internal.getMaxColumns() < 32) internal.insertColumnsAfter(internal.getMaxColumns(), 32 - internal.getMaxColumns());
    internal.getRange('A1').setValue('程式內部狀態，請勿編輯');
    const state = readState_(book) || newState();
    writeState_(book, state); refreshViews_(book, state); internal.hideSheet();
    SpreadsheetApp.flush();
    console.log('初始化完成。請部署為網頁應用程式，再把 /exec 網址填到本機 .env.local。');
  } finally { lock.releaseLock(); }
}
function readState_(book) {
  const sheet = book.getSheetByName('_State');
  if (!sheet || sheet.getMaxColumns() < 32) return null;
  const value = sheet.getRange(2, 1, 1, 32).getValues()[0].join('');
  return value ? JSON.parse(value) : null;
}
function writeState_(book, state) {
  const serialized = JSON.stringify(state);
  if (serialized.length > 32 * 40000) throw new Error('已超過練習版容量，請封存資料後再繼續');
  const chunks = Array.from({length:32}, (_, i) => serialized.slice(i * 40000, (i + 1) * 40000));
  // 同一範圍單次寫入完整狀態；投影表可重新建立。
  book.getSheetByName('_State').getRange(2, 1, 1, 32).setValues([chunks]);
  SpreadsheetApp.flush();
}
function literal_(value) { return typeof value === 'string' && /^[=+\-@]/.test(value) ? "'" + value : value; }
function writeView_(book, name, values) {
  const sheet = book.getSheetByName(name), width = SHEET_HEADERS[name].length;
  const oldRows = Math.max(sheet.getLastRow() - 1, 0), count = Math.max(oldRows, values.length);
  if (!count) return;
  if (sheet.getMaxRows() < count + 1) sheet.insertRowsAfter(sheet.getMaxRows(), count + 1 - sheet.getMaxRows());
  const rows = Array.from({length:count}, (_, i) => values[i] ? values[i].map(literal_) : Array(width).fill(''));
  sheet.getRange(2, 1, count, width).setValues(rows);
}
function refreshViews_(book, state) {
  writeView_(book, 'Records', state.records.map(row => { const r = row.record; return [row.date,row.shift,row.unit,r.status,r.staff,r.patients??'',r.住院??'',r.出院??'',r.轉入??'',r.轉出??'',r.病危??'',r.死亡??'',JSON.stringify(r),r.version,r.updatedAt]; }));
  writeView_(book, 'Openings', state.openings.map(r => [r.unit_id,r.report_date,r.shift,r.patients]));
  writeView_(book, 'Capacities', state.capacities.map(r => [r.unit,r.effectiveDate,r.beds,r.version,r.updatedAt||'',r.reason||'']));
  writeView_(book, 'Audit', state.audit.map(r => [r.created_at,r.action,r.target,'練習使用者',JSON.stringify(r.before),JSON.stringify(r.after)]));
}
function json_(value) { return ContentService.createTextOutput(JSON.stringify(value)).setMimeType(ContentService.MimeType.JSON); }
function doGet() { return json_({ service: 'NROP Google Sheets practice', version: '0.1.0' }); }
function doPost(e) {
  let lock;
  try {
    const cfg = settings_();
    if (!e || !e.postData || e.postData.contents.length > 40000) throw new Error('請求內容不正確');
    const input = JSON.parse(e.postData.contents);
    if (typeof input.token !== 'string' || input.token !== cfg.token) return json_({ error: { message: '讀寫服務憑證不正確', code: '42501' } });
    lock = LockService.getScriptLock(); lock.waitLock(20000);
    const book = SpreadsheetApp.openById(cfg.id), state = readState_(book);
    if (!state) throw new Error('請先在編輯器執行 setup');
    const params = Object.assign({}, input.params || {}, {p_requestId:Utilities.getUuid()});
    const result = transact(state, input.method, params);
    const mutation = ['nrop_save_record','nrop_set_opening','nrop_prepare_bed_change','nrop_confirm_bed_change'].includes(input.method);
    if (mutation) {
      writeState_(book, result.state);
      try { refreshViews_(book, result.state); SpreadsheetApp.flush(); }
      catch (viewError) { return json_({error:{message:'資料已儲存，但顯示分頁更新失敗。請在 Apps Script 重新執行 setup，勿直接重送。',code:'STORED'}}); }
    }
    return json_({data:result.data});
  } catch (error) { return json_({ error: { message: error.message || 'Google 試算表讀寫失敗', code: error.code || 'SHEETS_ERROR' } }); }
  finally { if (lock && lock.hasLock()) lock.releaseLock(); }
}

import { newState, transact, taiwanDate } from './data-model.js';
import { units, operationalUnits, shifts, admissionFieldsByUnit } from './catalog.js';
const DEMO_KEY = 'nrop-practice-demo-v1';
let mode = null;
export function selectMode(value) { if (!['demo', 'sheets'].includes(value)) throw new Error('模式不正確'); mode = value; }

export const storageLabel = () => mode === 'sheets' ? 'Google 試算表' : '本機示範（不會寫入 Google）';

  const response = await fetch('/api/connection', { signal: AbortSignal.timeout(10000) });
  if (!response.ok) throw new Error('本機資料服務未啟動，請使用 npm run dev 或 npm start。');
  return response.json();
}
function seed() {
  let state = newState(); const d = taiwanDate();
  units.forEach((u, i) => {
    state = transact(state, 'nrop_set_opening', { p_unit: u.id, p_date: d, p_shift: 'N', p_patients: Math.floor(u.beds * 0.65) }).state;
    shifts.forEach((s, j) => {
      const payload = { staff: 4 + i % 5, 住院: j === 1 ? 2 : 0, 出院: j === 2 ? 1 : 0, 轉入: 0, 轉出: 0, 病危: i % 3, 死亡: 0 };
      const split = admissionFieldsByUnit[u.id]; if (split) { payload[split[0][0]] = payload.住院; payload[split[1][0]] = 0; }
      state = transact(state, 'nrop_save_record', { p_unit: u.id, p_date: d, p_shift: s, p_payload: payload, p_status: 'submitted', p_version: 0 }).state;
    });
  });
  operationalUnits.forEach(u => shifts.forEach(s => {
    const payload = Object.fromEntries(u.fields.map(([f]) => [f, 0])); payload.staff = 5;
    if (u.type === 'nursingHome') { payload.nurseStaff = 3; payload.aideStaff = 4; payload.residents = 22; }
    state = transact(state, 'nrop_save_record', { p_unit: u.id, p_date: d, p_shift: s, p_payload: payload, p_status: 'submitted', p_version: 0 }).state;
  })); state.audit = []; return state;
}
export async function rpc(method, params = {}) {
  if (!mode) throw new Error('請先選擇練習模式');
  if (mode === 'demo') {
    if (!navigator.locks) throw new Error('請使用支援 Web Locks 的新版瀏覽器');
    return navigator.locks.request(DEMO_KEY, () => {
      const raw = localStorage.getItem(DEMO_KEY); let state;
      try { state = raw ? JSON.parse(raw) : seed(); } catch { throw new Error('示範資料無法讀取，請從入口重設示範資料。'); }
      const result = transact(state, method, { ...params, p_requestId: crypto.randomUUID() });
      localStorage.setItem(DEMO_KEY, JSON.stringify(result.state)); return result.data;
    });
  }
  const response = await fetch('/api/nrop', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ method, params }), signal: AbortSignal.timeout(50000) });
  const result = await response.json();
  if (result.error) throw Object.assign(new Error(result.error.message), { code: result.error.code });
  if (!response.ok) throw new Error('Google 試算表讀寫失敗'); return result.data;
}
export async function resetDemo() { await navigator.locks.request(DEMO_KEY, () => localStorage.removeItem(DEMO_KEY)); }

}

  if (Number(request.headers.get('content-length')) > 32000) throw new RangeError('資料超過大小限制');
  const reader = request.body?.getReader(); if (!reader) return '';
  const chunks = []; let total=0;
  while (true) { const {done,value}=await reader.read(); if(done)break; total+=value.byteLength; if(total>32000){await reader.cancel();throw new RangeError('資料超過大小限制');}chunks.push(value); }
  const bytes=new Uint8Array(total);let offset=0;for(const chunk of chunks){bytes.set(chunk,offset);offset+=chunk.byteLength;}return new TextDecoder().decode(bytes);

  if (!request.headers.get('oai-authenticated-user-id')) {
    if (path.startsWith('/api/')) return json(401,{error:{message:'請先使用網站擁有者的 ChatGPT 帳號登入。'}});
    return new Response('<!doctype html><html lang="zh-Hant"><meta charset="utf-8"><title>請登入</title><p>請登入以開啟私人練習工作區。</p><a href="/signin-with-chatgpt?return_to=/" target="_top">使用 ChatGPT 登入</a></html>',{status:401,headers:{...baseHeaders,'Content-Type':'text/html; charset=utf-8'}});
  }
  if (path.startsWith('/api/')) {
    if (request.headers.get('origin') && request.headers.get('origin')!==url.origin) return json(403,{error:{message:'僅允許本站操作'}});
    const configured=!!env.SHEETS_API_URL && typeof env.SHEETS_API_TOKEN==='string' && env.SHEETS_API_TOKEN.length>=32;
    if(path==='/api/connection' && request.method==='GET') return json(200,{configured,hosted:true,spreadsheetUrl:env.SHEETS_DOCUMENT_URL||''});
    if(path!=='/api/nrop') return json(404,{error:{message:'找不到資料服務'}});
    if(request.method!=='POST') return json(405,{error:{message:'不支援的請求方法'}});
    if(request.headers.get('origin')!==url.origin) return json(403,{error:{message:'僅允許本站操作'}});
    if(!configured || !/^https:\/\/script\.google\.com\/macros\/s\/[\w-]+\/exec$/.test(env.SHEETS_API_URL)) return json(503,{error:{message:'Google 讀寫服務尚未完成設定。'}});
    let input;
    try { input=JSON.parse(await inputText(request)); } catch(e) {return json(e instanceof RangeError?413:400,{error:{message:e instanceof RangeError?e.message:'請求內容不正確'}});}
    if(!input || !methods.has(input.method) || (input.params!=null && (typeof input.params!=='object' || Array.isArray(input.params)))) return json(400,{error:{message:'不支援的資料操作'}});
    try {
      const response=await remoteFetch(env.SHEETS_API_URL,{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({token:env.SHEETS_API_TOKEN,method:input.method,params:input.params||{}}),redirect:'follow',signal:AbortSignal.timeout(45000)});
      if(!response.ok) throw new Error('upstream');
      const result=await response.json();
      if(!result || (result.error && typeof result.error.message!=='string') || (!Object.hasOwn(result,'data')&&!result.error)) throw new Error('format');
      return json(200,result);
    }catch{return json(502,{error:{message:'Google 試算表連線中斷或回應異常，請重新整理確認儲存結果；系統不會自動重送。'}});}
  }
  if(!['GET','HEAD'].includes(request.method)) return new Response('不支援的請求方法',{status:405,headers:baseHeaders});
  const asset=assets[path==='/'?'/index.html':path];
  if(!asset)return new Response('找不到頁面',{status:404,headers:baseHeaders});
  const headers={...baseHeaders,'Content-Type':asset.type,'Content-Security-Policy':"default-src 'self'; script-src 'self'; style-src 'self' 'unsafe-inline'; img-src 'self' data:; connect-src 'self'; object-src 'none'; base-uri 'none'; form-action 'self'"};
  return new Response(request.method==='HEAD'?null:Uint8Array.from(atob(asset.data),c=>c.charCodeAt(0)),{status:200,headers});
}
